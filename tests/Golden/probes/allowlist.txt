The probe allowlist's corpus (V3_MILESTONES.md WP 1.4): each case is a line "=== admit: <what>" or "=== refuse: <what>"
and the probe text below it, up to the next case. Io.Tests' AllowlistTests read every case; ProbeTests run every admitted
case against the golden project as the read-only principal, so an admitted case names only the golden project's tables.

=== admit: COUNT of every row
SELECT COUNT(*) FROM dbo.Customer;
=== admit: COUNT_BIG of every row
SELECT COUNT_BIG(*) FROM dbo.Customer;
=== admit: COUNT of DISTINCT a column
SELECT COUNT(DISTINCT Region) FROM dbo.Customer;
=== admit: COUNT_BIG of DISTINCT a column named in two parts, the table aliased
SELECT COUNT_BIG(DISTINCT c.Region) FROM dbo.Customer AS c;
=== admit: SUM of CASE counting a column's nulls
SELECT SUM(CASE WHEN Email IS NULL THEN 1 ELSE 0 END) FROM dbo.Customer;
=== admit: SUM of CASE bucketing by length between literal boundaries
SELECT SUM(CASE WHEN LEN(Email) BETWEEN 1 AND 20 THEN 1 ELSE 0 END) AS short_emails FROM dbo.Customer;
=== admit: MIN over LEN
SELECT MIN(LEN(Name)) FROM dbo.Customer;
=== admit: MAX over LEN
SELECT MAX(LEN(Email)) FROM dbo.Customer;
=== admit: MIN over DATALENGTH
SELECT MIN(DATALENGTH(ContactPhone)) FROM dbo.Customer;
=== admit: MAX over DATALENGTH
SELECT MAX(DATALENGTH(Email)) FROM dbo.Customer;
=== admit: CASE WHEN EXISTS, the data-loss guard's predicate as DacFx writes it
SELECT CASE WHEN EXISTS (SELECT TOP 1 1 FROM [dbo].[Customer]) THEN 1 ELSE 0 END;
=== admit: an integer literal
SELECT 1;
=== admit: a negative integer literal
SELECT -1;
=== admit: several items, each aliased
SELECT COUNT_BIG(*) AS rows_total, SUM(CASE WHEN Email IS NULL THEN 1 ELSE 0 END) AS nulls, MAX(LEN(Email)) AS longest, 0 AS zero
FROM dbo.Customer;
=== admit: an item in parentheses
SELECT (COUNT_BIG(*)) FROM dbo.Status;
=== admit: orphans of a reference, counted through NOT EXISTS
SELECT COUNT_BIG(*) FROM dbo.Customer AS c
WHERE c.AccountId IS NOT NULL AND NOT EXISTS (SELECT 1 FROM dbo.Account AS a WHERE a.Id = c.AccountId);
=== admit: orphans of a reference, counted through a left join
SELECT SUM(CASE WHEN c.Id IS NULL THEN 1 ELSE 0 END) FROM dbo.[Order] AS o LEFT OUTER JOIN dbo.Customer AS c ON c.Id = o.CustomerId;
=== admit: an inner join with a literal filter
SELECT COUNT_BIG(*) FROM dbo.[Order] AS o INNER JOIN dbo.Customer AS c ON c.Id = o.CustomerId WHERE o.Total > 0;
=== admit: two columns compared with <>, the one inequality of two columns the allowlist holds
SELECT COUNT_BIG(*) FROM dbo.[Order] AS o INNER JOIN dbo.Customer AS c ON c.Id = o.CustomerId AND c.Id <> o.Id;
=== admit: values that would not convert, counted with TRY_CONVERT
SELECT SUM(CASE WHEN Code IS NOT NULL AND TRY_CONVERT(int, Code) IS NULL THEN 1 ELSE 0 END) FROM dbo.Product;
=== admit: TRY_CAST in a predicate
SELECT SUM(CASE WHEN TRY_CAST(Code AS nvarchar(10)) IS NULL THEN 1 ELSE 0 END) FROM dbo.Product;
=== admit: duplicates of a composite key, through a derived table
SELECT COUNT_BIG(*) FROM (SELECT OrderId, LineNumber FROM dbo.OrderLine GROUP BY OrderId, LineNumber HAVING COUNT_BIG(*) > 1) AS d;
=== admit: a length histogram, bucketed by length
SELECT MIN(LEN(Email)), COUNT_BIG(*) FROM dbo.Customer GROUP BY LEN(Email);
=== admit: untrusted foreign keys, read from the catalog
SELECT SUM(CASE WHEN is_not_trusted = 1 THEN 1 ELSE 0 END) FROM sys.foreign_keys;
=== admit: IN over literals and LIKE over a literal pattern
SELECT SUM(CASE WHEN Region IN (N'West', N'East') OR Email LIKE N'%@%' THEN 1 ELSE 0 END) FROM dbo.Customer;
=== admit: UPPER, ISNULL, COALESCE and NULLIF in a predicate
SELECT SUM(CASE WHEN UPPER(ISNULL(Region, N'')) = N'WEST' OR COALESCE(NULLIF(Email, N''), N'none') = N'none' THEN 1 ELSE 0 END) FROM dbo.Customer;
=== admit: IN over a subquery
SELECT COUNT_BIG(*) FROM dbo.Product WHERE CategoryId IN (SELECT Id FROM dbo.Category WHERE IsActive = 1);
=== admit: EXISTS nested in EXISTS, a column in the inner select list
SELECT CASE WHEN EXISTS (SELECT c.Email FROM dbo.Customer AS c WHERE EXISTS (SELECT 1 FROM dbo.[Order] AS o WHERE o.CustomerId = c.Id)) THEN 1 ELSE 0 END;
=== admit: a one-part name
SELECT COUNT(*) FROM Customer;
=== admit: comments, casing and whitespace around an allowed form
/* leading */ select
  count_big( * ) -- trailing
from   dbo.Customer ;
=== admit: one statement ended by a batch separator
SELECT COUNT(*) FROM dbo.Status;
GO

=== refuse: a join boundary read from the data, a column against another table's column

=== refuse: a join boundary read from the data through TRY_CONVERT

=== refuse: a join boundary read from the data through BETWEEN

=== refuse: a cross join boundary read from the data in SUM of CASE
SELECT SUM(CASE WHEN c.Id <= a.Balance THEN 1 ELSE 0 END) FROM dbo.Customer AS c CROSS JOIN dbo.Account AS a WHERE a.Id = 7;
=== refuse: a derived table's column as a length's boundary read from the data
SELECT COUNT_BIG(*) FROM dbo.Customer AS c CROSS JOIN (SELECT MAX(LEN(Email)) AS m FROM dbo.Customer) AS d WHERE LEN(c.Email) > d.m;
=== refuse: a correlated EXISTS boundary read from the data
SELECT COUNT_BIG(*) FROM dbo.Customer AS c WHERE EXISTS (SELECT 1 FROM dbo.Account AS a WHERE a.Id = 7 AND c.Id <= a.Balance);
=== refuse: an inequality of two columns written as an equality of their difference
SELECT SUM(CASE WHEN ABS(c.Id - a.Balance) + c.Id - a.Balance = 0 THEN 1 ELSE 0 END) FROM dbo.Customer AS c CROSS JOIN dbo.Account AS a WHERE a.Id = 7;
=== refuse: NULLIF of two columns, a boundary read from the data
SELECT SUM(CASE WHEN NULLIF(c.Id, a.Balance) IS NULL THEN 1 ELSE 0 END) FROM dbo.Customer AS c CROSS JOIN dbo.Account AS a WHERE a.Id = 7;
=== refuse: a value computed in a derived table's select list
SELECT SUM(CASE WHEN d.m <= 0 THEN 1 ELSE 0 END) FROM (SELECT c.Id - a.Balance AS m FROM dbo.Customer AS c CROSS JOIN dbo.Account AS a WHERE a.Id = 7) AS d;
=== refuse: a value computed in the select list of IN's subquery
SELECT COUNT_BIG(*) FROM dbo.Customer AS c WHERE 0 IN (SELECT ABS(c.Id - a.Balance) + c.Id - a.Balance FROM dbo.Account AS a WHERE a.Id = 7);
=== refuse: LIKE with a pattern read from the data

SELECT SUM(CASE WHEN LEN(c.Email) > LEN(c.Name) THEN 1 ELSE 0 END) FROM dbo.Customer AS c;
=== refuse: GROUP BY a column, buckets read from the data
