--- (
---     -- Explicit id, NO IDENTITY. Lookup rows own their keys.

---
---     CONSTRAINT PK_Status_Id PRIMARY KEY CLUSTERED (Id)
--- );
--- GO

--- -- One NULL is allowed under a UNIQUE index, so Code uniqueness is a clean separate proof.
